# ==============================================================================
# NEMATODE MEGA-PHYLOGENY: MACROEVOLUTIONARY ANALYSIS (v10 - EXTENDED, ROBUSTNESS & SENSITIVITY)
# ------------------------------------------------------------------------------
# This version addresses the following methodological issues identified in
# review of v8:
#
#   1. Imbalance Ratio (clade sizes) was reported as if it were a formal
#      statistical imbalance test. It is now reported alongside Colless and
#      Sackin indices (raw + normalized), plus a matched-subsampling analysis
#      that asks whether the imbalance is simply an artifact of very unequal
#      taxon sampling between the two super-clades.
#
#   2. Pybus & Harvey's Gamma statistic was interpreted as "definitive
#      evidence of Early Burst" without (a) checking whether the tree is
#      ultrametric, (b) testing significance against a proper null model,
#      and (c) considering that Gamma inflates with very large n. This
#      version adds: an ultrametricity check, an MCCR-style Monte Carlo
#      null-model significance test (Pybus & Harvey 2000), clade-specific
#      Gamma with its own null test, and a matched-subsampling Gamma
#      distribution.
#
#   3. Branch length (substitutions/site) was described as "molecular
#      evolutionary rate." This is only true if standardized by time. The
#      report language has been corrected throughout.
#
#   4. Sister-group claims were based purely on topological adjacency
#      without checking nodal support. This version only calls two clades
#      "sister groups" if the tree carries support values (bootstrap /
#      UFboot / posterior probability) at that node above a threshold;
#      otherwise it reports them as "topologically adjacent transition
#      zones" and says explicitly that no support values were available.
#
#   5. The host-jumping / cross-kingdom section previously wrote fixed,
#      unconditional prose regardless of whether guild data existed. It
#      is now conditional on a real guild/host column and only reports
#      descriptive co-occurrence, not causal "host-jumping" claims (which
#      would require ancestral state reconstruction of host use).
#
#   6. SIMMAP transition counts are now reported with mean, SD, and 95% CI
#      (not just a mean of 10 simulations), transition-rate asymmetry is
#      reported separately in each direction, and ER vs ARD models are
#      compared via AICc before choosing which model's root-state
#      probabilities to report (with the root probability given as a
#      distribution, not a single point estimate).
#
#   7. A "METHODOLOGICAL CAVEATS" block is printed at the top and bottom
#      of the report, listing checks that are NOT computable from a
#      pruned post-hoc tree alone (e.g., long-branch attraction, site-
#      heterogeneous substitution models, alignment/gene conflict,
#      geographic sampling bias, curation circularity) and therefore must
#      be addressed upstream in the alignment/tree-building pipeline.
#
# All original computations are preserved; nothing that worked before has
# been removed, only extended, corrected, or made conditional on evidence
# actually being available.
# ==============================================================================

# --- Colless index (raw + normalized), for fully bifurcating rooted trees ----
colless_index <- function(phy) {
  n <- Ntip(phy)
  if (n < 3) return(list(raw = NA, normalized = NA))
  internal_nodes <- (n + 1):(n + phy$Nnode)
  total <- 0
  for (node in internal_nodes) {
    children <- phy$edge[phy$edge[, 1] == node, 2]
    if (length(children) == 2) {
      l <- length(Descendants(phy, children[1], type = "tips")[[1]])
      r <- length(Descendants(phy, children[2], type = "tips")[[1]])
      total <- total + abs(l - r)
    }
  }
  norm <- total / ((n - 1) * (n - 2) / 2)
  list(raw = total, normalized = norm)
}

# --- Sackin index (raw + normalized), via node depth in edge counts ---------
sackin_index <- function(phy) {
  n <- Ntip(phy)
  phy_unweighted <- phy
  phy_unweighted$edge.length <- rep(1, nrow(phy$edge)) # اصلاح مقداردهی
  root <- n + 1
  depths <- dist.nodes(phy_unweighted)[root, 1:n]
  raw <- sum(depths)
  normalized <- raw / (n^1.5)   # Blum et al. (2006) style normalization
  list(raw = raw, normalized = normalized, depths = depths)
}

# --- MCCR-style Monte Carlo null-model test for Gamma (Pybus & Harvey 2000) --
# Simulates pure-birth (Yule) trees with the same number of tips (adjusted for
# sampling fraction rho), computes Gamma under the null, and derives a
# two-sided p-value and 95% CI for the null distribution.
mccr_gamma_test <- function(phy, rho = 1, n_sim = 1000) {
  n <- Ntip(phy)
  if (n < 3) return(list(obs_gamma = NA, n_sim_used = 0, p_value_one_tailed = NA, null_ci_2.5 = NA, null_ci_97.5 = NA, null_mean = NA))
  
  obs_gamma <- tryCatch(gammaStat(phy), error = function(e) NA_real_)
  if (is.na(obs_gamma)) return(list(obs_gamma = NA, n_sim_used = 0, p_value_one_tailed = NA, null_ci_2.5 = NA, null_ci_97.5 = NA, null_mean = NA))
  
  n_total <- max(n, ceiling(n / rho))
  sim_gammas <- rep(NA_real_, n_sim)
  for (i in seq_len(n_sim)) {
    sim_gammas[i] <- tryCatch({
      sim_tree <- rphylo(n = n_total, birth = 1, death = 0)
      if (rho < 1 && n_total > n) {
        keep <- sample(sim_tree$tip.label, n)
        sim_tree <- collapse.singles(keep.tip(sim_tree, keep))
      }
      if (Ntip(sim_tree) >= 3) gammaStat(sim_tree) else NA_real_
    }, error = function(e) NA_real_)
  }
  sim_gammas <- sim_gammas[!is.na(sim_gammas)]
  
  if (length(sim_gammas) == 0) return(list(obs_gamma = obs_gamma, n_sim_used = 0, p_value_one_tailed = NA, null_ci_2.5 = NA, null_ci_97.5 = NA, null_mean = NA))
  
  # Early Burst / early diversification predicts gamma << 0, so use a
  # one-tailed lower-tail p-value; also report the two-sided null CI.
  p_lower <- mean(sim_gammas <= obs_gamma)
  list(
    obs_gamma = obs_gamma,
    n_sim_used = length(sim_gammas),
    p_value_one_tailed = p_lower,
    null_ci_2.5 = as.numeric(quantile(sim_gammas, 0.025)),
    null_ci_97.5 = as.numeric(quantile(sim_gammas, 0.975)),
    null_mean = mean(sim_gammas)
  )
}

# --- Matched-subsampling: is the imbalance/Gamma driven by sample size? -----
matched_subsample_analysis <- function(phy, clade1_tips, clade2_tips, n_sim = 200) {
  size1 <- length(clade1_tips)
  size2 <- length(clade2_tips)
  min_size <- min(size1, size2)
  if (size1 >= size2) { larger <- clade1_tips; smaller <- clade2_tips
  } else { larger <- clade2_tips; smaller <- clade1_tips }
  
  gammas <- rep(NA_real_, n_sim)
  ratios <- rep(NA_real_, n_sim)
  for (i in seq_len(n_sim)) {
    if (min_size < 3) next
    sub_larger <- sample(larger, min_size)
    sub_tips <- c(sub_larger, smaller)
    sub_tree <- tryCatch(collapse.singles(keep.tip(phy, sub_tips)), error = function(e) NULL)
    if (!is.null(sub_tree) && Ntip(sub_tree) >= 3) {
      gammas[i] <- tryCatch(gammaStat(sub_tree), error = function(e) NA_real_)
    }
    ratios[i] <- 1  # by construction, matched subsampling forces ratio = 1
  }
  gammas <- gammas[!is.na(gammas)]
  
  if(length(gammas) == 0) {
    return(list(min_size = min_size, n_sim_used = 0, gamma_mean = NA, gamma_sd = NA, gamma_ci_2.5 = NA, gamma_ci_97.5 = NA))
  }
  
  list(
    min_size = min_size,
    n_sim_used = length(gammas),
    gamma_mean = mean(gammas),
    gamma_sd = sd(gammas),
    gamma_ci_2.5 = as.numeric(quantile(gammas, 0.025)),
    gamma_ci_97.5 = as.numeric(quantile(gammas, 0.975))
  )
}

# --- AICc for discrete-trait models (ER vs ARD) -------------------------------
aicc_discrete <- function(loglik, k, n) {
  -2 * loglik + 2 * k + (2 * k * (k + 1)) / (n - k - 1)
}

          paste(gen_left, "(Parasitic) <->", gen_right, "(Free-living)")
        } else {
          paste(gen_right, "(Parasitic) <->", gen_left, "(Free-living)")
        }
        
        if (!is.na(support_val) && support_val >= SUPPORT_THRESHOLD) {
          sister_transitions <- c(sister_transitions,
                                  sprintf("%s [support = %.1f]", label_txt, support_val))
        } else {
          adjacent_zones <- c(adjacent_zones,
                              sprintf("%s [support = %s]", label_txt,
                                      if (is.na(support_val)) "not available" else sprintf("%.1f (below threshold)", support_val)))
        }
      }
    }
  }
}

# --- Robust AICc helpers -------------------------------------------------------
delta_aicc <- function(aicc_values) {
  aicc_values <- aicc_values[is.finite(aicc_values)]
  if (!length(aicc_values)) return(numeric(0))
  aicc_values - min(aicc_values)
}
akaike_weights <- function(aicc_values) {
  if (!length(aicc_values)) return(numeric(0))
  aa <- aicc_values
  ok <- is.finite(aa)
  w <- rep(NA_real_, length(aa))
  if (any(ok)) {
    d <- aa[ok] - min(aa[ok])
    z <- exp(-0.5 * d)
    w[ok] <- z / sum(z)
  }
  w
}

# --- Stratified rarefaction: preserves lifestyle composition ------------------
stratified_rarefaction <- function(phy, state_vec, fractions = c(0.75, 0.5, 0.25), nrep = 200) {
  state_vec <- state_vec[phy$tip.label]
  out <- list()
  for (f in fractions) {
    vals <- data.frame(gamma = numeric(0), colless = numeric(0), sackin = numeric(0))
    for (r in seq_len(nrep)) {
      keep <- unlist(lapply(split(names(state_vec), state_vec), function(x) {
        k <- max(2, floor(length(x) * f))
        if (length(x) <= k) x else sample(x, k)
      }))
      if (length(keep) < 5) next
      st <- tryCatch(collapse.singles(keep.tip(phy, keep)), error = function(e) NULL)
      if (is.null(st)) next
      gg <- tryCatch(if(Ntip(st)>=3) gammaStat(st) else NA_real_, error = function(e) NA_real_)
      cc <- tryCatch(colless_index(st)$normalized, error = function(e) NA_real_)
      ss <- tryCatch(sackin_index(st)$normalized, error = function(e) NA_real_)
      vals <- rbind(vals, data.frame(gamma = gg, colless = cc, sackin = ss))
    }
    out[[as.character(f)]] <- vals
  }
  out
}

# --- Formal best-model adequacy simulation ------------------------------------
# If RPANDA has produced a time-dependent best model, simulate under that fitted
# time-dependent rate function. Otherwise retain the original Yule/BD adequacy
# simulation. This is an additive extension; the original calculations remain.
simulate_under_rpanda_best <- function(phy, rp_fits, n_sim = 500) {
  if (is.null(rp_fits) || is.null(rp_fits$fits)) return(list())
  best_name <- names(which.min(rp_fits$aicc))
  fit <- rp_fits$fits[[best_name]]
  tot_time <- rp_fits$tot_time
  f <- rp_fits$sampling_fraction
  out <- vector("list", n_sim)
  
  if (best_name == "Constant_PureBirth") {
    lambda <- as.numeric(fit$lamb_par[1])
    for (i in seq_len(n_sim)) {
      out[[i]] <- tryCatch(
        collapse.singles(ape::rphylo(n=Ntip(phy), birth=lambda, death=0, T0=tot_time)),
        error=function(e) NULL
      )
    }
  } else if (best_name == "Exponential_PureBirth") {
    lambda0 <- as.numeric(fit$lamb_par[1])
    beta <- as.numeric(fit$lamb_par[2])
    birth_fun <- function(t) lambda0 * exp(beta * t)
    for (i in seq_len(n_sim)) {
      out[[i]] <- tryCatch(
        collapse.singles(ape::rphylo(n=Ntip(phy), birth=birth_fun, death=0, T0=tot_time)),
        error=function(e) NULL
      )
    }
  } else if (best_name == "Exponential_BirthDeath") {
    lambda0 <- as.numeric(fit$lamb_par[1])
    beta <- as.numeric(fit$lamb_par[2])
    mu0 <- as.numeric(fit$mu_par[1])
    birth_fun <- function(t) lambda0 * exp(beta * t)
    death_fun <- function(t) pmax(0, mu0)
    for (i in seq_len(n_sim)) {
      out[[i]] <- tryCatch(
        collapse.singles(ape::rphylo(n=Ntip(phy), birth=birth_fun, death=death_fun, T0=tot_time)),
        error=function(e) NULL
      )
    }
  }
  out[!sapply(out, is.null)]
}

best_model_simulations <- NULL
if (exists("rpanda_fits") && !is.null(rpanda_fits) && !is.null(analysis_tree_time)) {
  best_model_simulations <- simulate_under_rpanda_best(analysis_tree_time, rpanda_fits, N_SIM_MODEL_ADEQUACY)
}

# --- Formal time-dependent diversification models via RPANDA ------------------
# RPANDA is used only with a validated ultrametric/time-calibrated tree.
# The exponential speciation-rate model is the formal Early-Burst / declining-
# through-time-rate test; t is measured backward from the present.
RUN_RPANDA <- TRUE
rpanda_fits <- NULL
rpanda_result <- NULL
if (RUN_RPANDA && !is.null(analysis_tree_time) && requireNamespace("RPANDA", quietly = TRUE)) {
  rpanda_result <- tryCatch({
    tot_time <- max(ape::node.age(analysis_tree_time)$ages, na.rm = TRUE)
    f_use <- RHO_SAMPLING_FRACTION
    if (!is.finite(f_use) || f_use <= 0 || f_use > 1) f_use <- 1
    
    # Model 1: constant pure-birth
    f.lamb.cst <- function(t, y) y[1]
    f.mu.zero <- function(t, y) 0
    fit_cst <- RPANDA::fit_bd(
      analysis_tree_time, tot_time, f.lamb.cst, f.mu.zero,
      lamb_par = c(max(0.001, 1 / tot_time)), mu_par = c(),
      f = f_use, cst.lamb = TRUE, fix.mu = TRUE, dt = 1e-3
    )
    fit_cst$model_label <- "Constant pure-birth"
    
    # Model 2: exponential time-varying speciation, no extinction.
    # lambda(t) = lambda0 * exp(beta * t); beta > 0 means higher rates in the past
    # and therefore a decline toward the present (Early-Burst-like pattern).
    f.lamb.exp <- function(t, y) y[1] * exp(y[2] * t)
    fit_exp <- RPANDA::fit_bd(
      analysis_tree_time, tot_time, f.lamb.exp, f.mu.zero,
      lamb_par = c(max(0.001, 1 / tot_time), 0.01), mu_par = c(),
      f = f_use, expo.lamb = TRUE, fix.mu = TRUE, dt = 1e-3
    )
    fit_exp$model_label <- "Exponential time-varying pure-birth (Early-Burst test)"
    
    # Model 3: exponential speciation + constant extinction.
    f.mu.cst <- function(t, y) y[1]
    fit_exp_bd <- RPANDA::fit_bd(
      analysis_tree_time, tot_time, f.lamb.exp, f.mu.cst,
      lamb_par = c(max(0.001, 1 / tot_time), 0.01), mu_par = c(max(1e-5, 0.1 / tot_time)),
      f = f_use, expo.lamb = TRUE, cst.mu = TRUE, dt = 1e-3
    )
    fit_exp_bd$model_label <- "Exponential time-varying birth-death"
    
    fits <- list(Constant_PureBirth = fit_cst, Exponential_PureBirth = fit_exp, Exponential_BirthDeath = fit_exp_bd)
    aicc_vals <- sapply(fits, function(x) x$aicc)
    dAICc <- aicc_vals - min(aicc_vals)
    weights <- exp(-0.5 * dAICc) / sum(exp(-0.5 * dAICc))
    rpanda_fits <<- list(fits=fits, aicc=aicc_vals, delta_aicc=dAICc, weights=weights, tot_time=tot_time, sampling_fraction=f_use)
    
    list(available=TRUE, note=sprintf("RPANDA fit_bd completed on a validated time tree; best model = %s.", names(which.min(aicc_vals))), fits=rpanda_fits)
  }, error=function(e) list(available=FALSE, note=paste("RPANDA fit failed:", conditionMessage(e))))
} else if (RUN_RPANDA) {
  rpanda_result <- list(available=FALSE, note="RPANDA model fitting skipped: a validated ultrametric/time-calibrated tree and/or RPANDA package was unavailable.")
}

# LTT Plot (High-Res PDF)
if (exists("ltt_tree") && !is.null(ltt_tree)) {
  pdf(file.path(out_dir, "LTT_Plot_Publication_Quality.pdf"), width = 8, height = 6)
  
  if (exists("null_trees") && length(null_trees) > 0) {
    
    # 1. Safely filter out any NULL/invalid elements from the null_trees list
    if (inherits(null_trees, "phylo")) {
      null_list <- list(null_trees)
    } else {
      null_list <- Filter(function(x) inherits(x, "phylo"), null_trees)
    }
    
    if (length(null_list) > 0) {
      # 2. Plot the empirical tree first to initialize the plot window and axes
      ape::ltt.plot(ltt_tree, backward = TRUE, col = "red", lwd = 3,
                    main = "Lineage-Through-Time (LTT) Plot with Null Envelope")
      
      # 3. Add all valid null trees as background lines
      for (i in seq_along(null_list)) {
        ape::ltt.lines(null_list[[i]], backward = TRUE, col = "grey80", lwd = 0.5)
      }
      
      # 4. Redraw the empirical tree ON TOP so it isn't buried behind the grey lines
      ape::ltt.lines(ltt_tree, backward = TRUE, col = "red", lwd = 3)
      
    } else {
      # Fallback if no valid null trees are found
      ape::ltt.plot(ltt_tree, backward = TRUE, col = "red", lwd = 3, 
                    main = "Lineage-Through-Time (LTT) Plot")
    }
    
  } else {
    ape::ltt.plot(ltt_tree, backward = TRUE, col = "red", lwd = 3, 
                  main = "Lineage-Through-Time (LTT) Plot")
  }
  
  dev.off()
}
